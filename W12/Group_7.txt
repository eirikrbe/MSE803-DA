import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.ensemble import RandomForestClassifier
from sklearn.feature_selection import mutual_info_classif
from sklearn.metrics import accuracy_score, classification_report

# 1. Load the Group 7 Excel file
df = pd.read_excel('Group_7.xlsx')

# Define all features and target
X_all = df[['SL', 'SW', 'PL', 'PW']]
y = df['Species']

print("==================================================")
print(" GROUP 7: FEATURE INFORMATION GAIN & HYPOTHESIS TEST")
print("==================================================")

# 2. Train Random Forest and calculate Information Gain (Mutual Information)
rf_full = RandomForestClassifier(n_estimators=100, random_state=42)
rf_full.fit(X_all, y)
mi_scores = mutual_info_classif(X_all, y)

# 3. Create Feature Importance Summary Table
importance_df = pd.DataFrame({
    'Feature': X_all.columns,
    'Random_Forest_Importance': rf_full.feature_importances_,
    'Information_Gain_Score': mi_scores
}).sort_values(by='Random_Forest_Importance', ascending=False)

print("\n--- Feature Importance Rankings ---")
print(importance_df)

# Aggregate group importances to test H1 vs H0
petal_importance = importance_df[importance_df['Feature'].isin(['PL', 'PW'])]['Random_Forest_Importance'].sum()
sepal_importance = importance_df[importance_df['Feature'].isin(['SL', 'SW'])]['Random_Forest_Importance'].sum()

print(f"\nTotal Petal Features Importance (PL + PW): {petal_importance * 100:.2f}%")
print(f"Total Sepal Features Importance (SL + SW): {sepal_importance * 100:.2f}%")

if petal_importance > sepal_importance:
    print("\n[VERDICT]: H1 is supported. Petal features carry more information than sepal features.")
else:
    print("\n[VERDICT]: H0 holds.")

# 4. Group 7 Specific Sepal ML Model Evaluation
X_sepal = df[['SL', 'SW']]
rf_sepal = RandomForestClassifier(n_estimators=100, random_state=42)
rf_sepal.fit(X_sepal, y)
df['Pred_Group7_Python'] = rf_sepal.predict(X_sepal)

print(f"\n--> Group 7 Sepal-Only ML Accuracy: {accuracy_score(y, df['Pred_Group7_Python']) * 100:.2f}%")
print("\n--- Sepal Classification Report ---")
print(classification_report(y, df['Pred_Group7_Python']))

# 5. Scatter Plot Visualization for Group 7
plt.figure(figsize=(8, 6))
sns.scatterplot(data=df, x='SL', y='SW', hue='Species', palette='Set2', s=80)
plt.title('Group 7: Sepal Dimensions (SL vs SW) Clustering')
plt.xlabel('Sepal Length (SL) [cm]')
plt.ylabel('Sepal Width (SW) [cm]')
plt.grid(True, linestyle='--', alpha=0.5)
plt.show()

# Export finalized sheet
df.to_excel('Group_7_Final_Completed.xlsx', index=False)
